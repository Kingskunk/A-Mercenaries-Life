*comment THE DEFENSE CORE (2026-10-31, the user: any defender, any number of types, as close to RAW as possible). defense_flags reads one defender's three damage-type lists, written as words with spaces between ("fire poison cold"), for one
*comment damage type, by the defender's field PREFIX: "combat_enemy_" (the active target's scratch), "combat_enemy3_" (one slot), "combat_ally2_" or "player_". It sets combat_def_immune, combat_def_resist, combat_def_vulnerable and, when the defender has

*comment and "combat_ally0..4_" (NEVER on combat_enemy_ scratch). The existing ten-round maximums stay as each spell's
*comment ceiling; concentration adds two earlier ways for effects to drop: casting another concentration spell
*comment (concentration_begin) or failing a Constitution saving throw on taking combat damage
*comment (concentration_damage_check: DC max(10, floor(damage / 2)); Arcane Ward absorption is excluded, temp HP counts,
*comment dropping to 0 HP makes no save). Dying also calls concentration_end. Damage checks are wired for the player
*comment in resolve_enemy_attack; enemy/ally callers are deferred.
*comment How to add a new concentration spell:
*comment   1. Add "concentration": true to its record in tools/spell_book.json and run `node tools/gen_spell_book.js`.
*comment   2. Add an explicit `*if (ce_id = "<spell_id>")` branch in concentration_end below to clear all ongoing combat
*comment      state the spell leaves behind (explicit branches only, never construct variable names from spell_id), and add
*comment      its display name to concentration_format_ended_banner.
*comment   3. If the spell ends on round expiry or when its targets clear/die, add its active check to
*comment      concentration_reconcile below so concentration returns to "none" without needing another cast.
*comment   4. Inside the cast's replay lock (combat_spells.txt / combat.txt), call
*comment      `*gosub_scene combat_dice concentration_begin "<prefix>" "<spell_id>"` (and clear via concentration_end if
*comment      zero targets failed the initial save); outside the lock, call
*comment      `*gosub_scene combat_prose print_concentration_ended_banner`.
*comment ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

*label concentration_has
*params chs_prefix
*temp chs_val {(chs_prefix & "concentrating")}
*set combat_conc_result (not(chs_val = "none"))
*return

*label concentration_end
*params ce_prefix
*temp ce_id {(ce_prefix & "concentrating")}
*if (ce_id = "none")
  *return
*set {(ce_prefix & "concentrating")} "none"
*if (ce_id = "hex")
  *if (ce_prefix = "combat_player_")
    *set hex_active false
    *set hex_target_slot 0
  *return
*if (ce_id = "hunters_mark")
  *if (ce_prefix = "combat_player_")
    *set favored_foe_target_slot 0
    *set favored_foe_expires_round 0
  *return
*if (ce_id = "ensnaring_strike")
  *if (ce_prefix = "combat_player_")
    *set ensnaring_armed false
    *set ensnaring_expires_round 0
  *comment Only the restraint this caster's Ensnaring Strike put on.
  *gosub condition_end_by_source "restrained" (ce_prefix & "ensnaring_strike")
  *return
*if (ce_id = "bane")
  *if (ce_prefix = "combat_player_")
    *temp ce_bi 1
    *label ce_bane_enemy_loop
    *if (ce_bi <= combat_enemy_slot_cap)
      *set {(("combat_enemy" & ce_bi) & "_bane_expires_round")} 0
      *set ce_bi + 1
      *goto ce_bane_enemy_loop
  *else
    *set combat_player_bane_expires_round 0
    *temp ce_bai 1
    *label ce_bane_ally_loop
    *if (ce_bai <= combat_ally_slot_cap)
      *set {(("combat_ally" & ce_bai) & "_bane_expires_round")} 0
      *set ce_bai + 1
      *goto ce_bane_ally_loop
  *return
*comment Only the fear this caster's Cause Fear put on: a creature frightened by anything else stays frightened.
*if (ce_id = "cause_fear")
  *gosub condition_end_by_source "frightened" (ce_prefix & "cause_fear")
  *return
*comment Only the hold this caster's Hold Person put on.
*if (ce_id = "hold_person")
  *gosub condition_end_by_source "paralyzed" (ce_prefix & "hold_person")
  *return
*if (ce_id = "blade_ward")
  *if (ce_prefix = "combat_player_")
    *set blade_ward_rounds 0
    *set blade_ward_just_cast false
  *return
*return

*label concentration_format_ended_banner
*params cf_prefix cf_spell_id
*temp cf_name cf_spell_id
*if (cf_spell_id = "hex")
  *set cf_name "Hex"
*if (cf_spell_id = "hunters_mark")
  *set cf_name "Hunter's Mark"
*if (cf_spell_id = "ensnaring_strike")
  *set cf_name "Ensnaring Strike"
*if (cf_spell_id = "bane")
  *set cf_name "Bane"
*if (cf_spell_id = "cause_fear")
  *set cf_name "Cause Fear"
*if (cf_spell_id = "hold_person")
  *set cf_name "Hold Person"
*if (cf_spell_id = "blade_ward")
  *set cf_name "Blade Ward"
*if (cf_prefix = "combat_player_")
  *set combat_conc_ended_banner "[b][🔮 Concentration on ${cf_name} ends][/b]"
*else
  *temp cf_cname {(cf_prefix & "name")}
  *set combat_conc_ended_banner "[b][🔮 ${cf_cname}: concentration on ${cf_name} ends][/b]"
*return

*label concentration_begin
*params cb_prefix cb_spell_id
*set combat_conc_ended_banner ""
*temp cb_prev {(cb_prefix & "concentrating")}
*if (not(cb_prev = "none"))
  *gosub concentration_end cb_prefix
  *if (not(cb_prev = cb_spell_id))
    *gosub concentration_format_ended_banner cb_prefix cb_prev
*set {(cb_prefix & "concentrating")} cb_spell_id
*return

*comment Incapacitated (Stunned and the like) ends concentration with no save. Builds the same "ends" banner the damage save does; the caller prints it.
*comment Wired for the player's stun in resolve_enemy_attack. An enemy or ally that concentrates calls this wherever it is stunned.
*label concentration_end_incapacitated
*params cei_prefix
*temp cei_spell {(cei_prefix & "concentrating")}
*if (cei_spell = "none")
  *return
*gosub concentration_end cei_prefix
*gosub concentration_format_ended_banner cei_prefix cei_spell
*return

*label concentration_damage_check
*params cdc_prefix cdc_dmg cdc_lock
*if (cdc_dmg <= 0)
  *return
*temp cdc_spell {(cdc_prefix & "concentrating")}
*if (cdc_spell = "none")
  *return
*comment Future enemy/ally callers guard {prefix}hp > 0 upstream (player uses hp_current, not combat_player_hp).
*if ((cdc_prefix = "combat_player_") and (hp_current <= 0))
  *return
*comment A non-lethal fight stops the player at 1 HP and ends there as a rescue (combat_next_turn), so that blow is the same as dropping to 0: no save.
*if (((cdc_prefix = "combat_player_") and (not(combat_player_can_die))) and (hp_current <= 1))
  *return
*temp cdc_half ((cdc_dmg - (cdc_dmg modulo 2)) / 2)
*temp cdc_dc 10
*if (cdc_half > 10)
  *set cdc_dc cdc_half
*set combat_save_dmg_type ""
*comment Feat hook: War Caster would grant advantage on this concentration Constitution save.
*set {("combat_save_locked_" & cdc_lock)} false
*gosub resolve_save cdc_prefix "constitution" "Constitution" cdc_dc cdc_lock
*if (cdc_prefix = "combat_player_")
  *set combat_conc_save_triggered true
  *set combat_conc_save_banner {("combat_save_banner_" & cdc_lock)}
*if (check_success)
  *gosub concentration_end cdc_prefix
  *gosub concentration_format_ended_banner cdc_prefix cdc_spell
*return

*label concentration_reconcile
*if (combat_player_concentrating = "none")
  *return
*if (combat_player_concentrating = "blade_ward")
  *if (blade_ward_rounds <= 0)
    *gosub concentration_end "combat_player_"
  *return
*if (combat_player_concentrating = "hex")
  *if ((not(hex_active)) or (hex_target_slot <= 0))
    *gosub concentration_end "combat_player_"
  *return
*if (combat_player_concentrating = "hunters_mark")
  *if ((favored_foe_target_slot <= 0) or (combat_round > favored_foe_expires_round))
    *gosub concentration_end "combat_player_"
  *return
*if (combat_player_concentrating = "ensnaring_strike")
  *if ((ensnaring_armed) and (ensnaring_expires_round >= combat_round))
    *return
  *temp cre_i 1
  *label cre_ensnare_loop
  *if (cre_i <= combat_enemy_count)
    *temp cre_hp {(("combat_enemy" & cre_i) & "_hp")}
    *temp cre_r {(("combat_enemy" & cre_i) & "_restrained")}
    *temp cre_src {(("combat_enemy" & cre_i) & "_restrained_src")}
    *comment The spell's last round (ensnaring_expires_round) is its 10-round ceiling: after it, concentration ends here and the restraint with it.
    *if ((((cre_hp > 0) and (cre_r > 0)) and (cre_src = "combat_player_ensnaring_strike")) and (ensnaring_expires_round >= combat_round))
      *return
    *set cre_i + 1
    *goto cre_ensnare_loop
  *gosub concentration_end "combat_player_"
  *return
*if (combat_player_concentrating = "bane")
  *temp crb_i 1
  *label crb_bane_loop
  *if (crb_i <= combat_enemy_count)
    *temp crb_hp {(("combat_enemy" & crb_i) & "_hp")}
    *temp crb_be {(("combat_enemy" & crb_i) & "_bane_expires_round")}
    *if ((crb_hp > 0) and (crb_be >= combat_round))
      *return
    *set crb_i + 1
    *goto crb_bane_loop
  *gosub concentration_end "combat_player_"
  *return
*if (combat_player_concentrating = "cause_fear")
  *temp crf_i 1
  *label crf_fear_loop
  *if (crf_i <= combat_enemy_count)
    *temp crf_hp {(("combat_enemy" & crf_i) & "_hp")}
    *temp crf_fr {(("combat_enemy" & crf_i) & "_frightened")}
    *temp crf_src {(("combat_enemy" & crf_i) & "_frightened_src")}
    *if (((crf_hp > 0) and (crf_fr > 0)) and (crf_src = "combat_player_cause_fear"))
      *return
    *set crf_i + 1
    *goto crf_fear_loop
  *gosub concentration_end "combat_player_"
  *return
*comment Hold Person ends once no living foe is still held by it: the hold broken by a save, its 10 rounds run out, or the held foe dead.
*if (combat_player_concentrating = "hold_person")
  *temp crh_i 1
  *label crh_hold_loop
  *if (crh_i <= combat_enemy_count)
    *temp crh_hp {(("combat_enemy" & crh_i) & "_hp")}
    *temp crh_pa {(("combat_enemy" & crh_i) & "_paralyzed")}
    *temp crh_src {(("combat_enemy" & crh_i) & "_paralyzed_src")}
    *if (((crh_hp > 0) and (crh_pa > 0)) and (crh_src = "combat_player_hold_person"))
      *return
    *set crh_i + 1
    *goto crh_hold_loop
  *gosub concentration_end "combat_player_"
  *return
