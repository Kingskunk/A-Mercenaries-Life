*label resolve_enemy_save
*params res_prefix res_stat res_stat_label res_lock_index
*temp res_lockname ("combat_save_locked_" & res_lock_index)
*temp res_bannername ("combat_save_banner_" & res_lock_index)
*temp res_successname ("combat_save_success_" & res_lock_index)
*temp res_banner ""
*if ({(res_lockname)}) and (locked_combat_save_page_id = page_id)
  *set check_success {(res_successname)}
  *set res_banner {(res_bannername)}
*else
  *temp res_score {(res_prefix & res_stat)}
  *gosub_scene startup score_to_mod res_score
  *temp res_mod sm_mod
  *temp res_roll 0
  *rand res_roll 1 20
  *temp res_total (res_roll + res_mod)
  *temp res_bane_str ""
  *temp res_bane_expires {(res_prefix & "bane_expires_round")}
  *if (res_bane_expires >= combat_round)
    *temp res_bane_die 0
    *rand res_bane_die 1 4
    *set res_total - res_bane_die
    *set res_bane_str " -${res_bane_die} (Bane)"
  *set check_success (res_total < spell_save_dc)
  *temp res_mod_str ""
  *if (res_mod >= 0)
    *set res_mod_str "+${res_mod}"
  *else
    *set res_mod_str "${res_mod}"
  *temp res_name {(res_prefix & "name")}
  *temp res_verdict "SAVED"
  *if (check_success)
    *set res_verdict "FAILED"
  *set res_banner "[b][🎲 ${res_name} Saving Throw: ${res_stat_label} (DC ${spell_save_dc}) | Roll: ${res_roll} ${res_mod_str}${res_bane_str} = ${res_total} — ${res_verdict}][/b]"
  *set {(res_successname)} check_success
  *set {(res_bannername)} res_banner
  *set {(res_lockname)} true
  *set locked_combat_save_page_id page_id
${res_banner}
