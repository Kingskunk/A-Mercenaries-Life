*comment *gosub_scene equipment format_copper <amount>   (output gear_sale_desc, e.g. "3 Silver Marks, 5 Copper Bits")
*label format_copper
*params fc_amount
*temp fc_silver ((fc_amount - (fc_amount modulo 10)) / 10)
*temp fc_copper (fc_amount modulo 10)
*temp fc_s_str ""
*temp fc_c_str ""
*if (fc_silver = 1)
  *set fc_s_str "1 Silver Mark"
*elseif (fc_silver > 1)
  *set fc_s_str (fc_silver & " Silver Marks")
*if (fc_copper = 1)
  *set fc_c_str "1 Copper Bit"
*elseif (fc_copper > 1)
  *set fc_c_str (fc_copper & " Copper Bits")
*if ((fc_silver > 0) and (fc_copper > 0))
  *set gear_sale_desc ((fc_s_str & ", ") & fc_c_str)
*elseif (fc_silver > 0)
  *set gear_sale_desc fc_s_str
*elseif (fc_copper > 0)
  *set gear_sale_desc fc_c_str
*else
  *set gear_sale_desc "nothing"
*return

*comment *gosub_scene equipment gear_unit_value <item_id> <"smith"|"pawn"|"general"|"tailor">   (output gear_unit_copper: what ONE
*comment unit fetches from that buyer; 0 means this buyer declines it)
*comment Prices come from the catalog's sell fields (gear_info): a percentage of retail per buyer, an optional fixed
*comment pawn value, and an optional per-piece smith price for a counted item. "general" is the general store.
*label gear_unit_value
*params guv_item guv_buyer
*gosub gear_info guv_item
*temp guv_pct gear_pct_pawn
*if (guv_buyer = "smith")
  *set guv_pct gear_pct_smith
*if (guv_buyer = "general")
  *set guv_pct gear_pct_general
*if (guv_buyer = "tailor")
  *set guv_pct gear_pct_tailor
*set gear_unit_copper (((gear_retail * guv_pct) - ((gear_retail * guv_pct) modulo 100)) / 100)
*if ((guv_buyer = "pawn") and (gear_fixed_pawn > 0))
  *set gear_unit_copper gear_fixed_pawn
*if ((guv_buyer = "smith") and (gear_piece_smith > 0))
  *set gear_unit_copper gear_piece_smith
*return

*comment *gosub_scene equipment gear_sale_value <item_id> <"smith"|"pawn"|"general"|"tailor">   (a price of 0 means this buyer declines)
*comment The whole-lot price the shop menus quote: one unit, or every piece for a counted item like scrap steel.
*label gear_sale_value
*params gsv_item gsv_buyer
*gosub gear_unit_value gsv_item gsv_buyer
*set gear_sale_name gear_name
*set gear_sale_copper gear_unit_copper
*if ((gsv_buyer = "smith") and (gear_piece_smith > 0))
  *set gear_sale_copper (gear_piece_smith * {gear_count_var})
*gosub format_copper gear_sale_copper
*return
