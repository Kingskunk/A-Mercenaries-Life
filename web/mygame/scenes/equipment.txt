*comment *gosub_scene equipment format_copper <amount>   (output gear_sale_desc, e.g. "3 Silver Marks, 5 Copper Bits")
*label format_copper
*params fc_amount
*temp fc_silver ((fc_amount - (fc_amount modulo 10)) / 10)
*temp fc_copper (fc_amount modulo 10)
*temp fc_s_str ""
*temp fc_c_str ""
*if (fc_silver = 1)
  *set fc_s_str "1 Silver Mark"
*elseif (fc_silver > 1)
  *set fc_s_str (fc_silver & " Silver Marks")
*if (fc_copper = 1)
  *set fc_c_str "1 Copper Bit"
*elseif (fc_copper > 1)
  *set fc_c_str (fc_copper & " Copper Bits")
*if ((fc_silver > 0) and (fc_copper > 0))
  *set gear_sale_desc ((fc_s_str & ", ") & fc_c_str)
*elseif (fc_silver > 0)
  *set gear_sale_desc fc_s_str
*elseif (fc_copper > 0)
  *set gear_sale_desc fc_c_str
*else
  *set gear_sale_desc "nothing"
*return
