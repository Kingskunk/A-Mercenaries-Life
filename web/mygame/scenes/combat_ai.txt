  *if (((not(fpais_id = "")) and (not(fpais_id = {(fpais_p & "spell_base_id")}))) and ((fpais_uses > 0) and (fpais_chance > 0)))
    *gosub spell_worth_casting_at_ally fpais_p fpais_id fpais_ally
    *if (((combat_ai_trace) and (not(combat_spell_ff_note = ""))) and (not(combat_spell_worth_it)))
      *gosub ai_trace {(fpais_p & "name")} "spell at an ally: ${fpais_id} ${combat_spell_ff_note}not worth casting"
    *if (combat_spell_worth_it)
